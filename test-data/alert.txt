import test, { expect } from "@playwright/test";
test("handling alert",async({page})=>{
    await page.goto('https://the-internet.herokuapp.com/javascript_alerts')
    await page.waitForTimeout(3000)
    //create dialog listerner
    page.on('dialog',async (dialog)=>{
        //return type of alert
        const alertytpe = dialog.type()
        console.log(`Type Of Alert is::${alertytpe}`)
        //return alert message
        const alertMessage = dialog.message()
        console.log(`Alert Message is::${alertMessage}`)
        await page.waitForTimeout(4000)
        //click ok in alert
        dialog.accept()
    })
    await page.getByText('Click for JS Alert').click()
    await page.waitForTimeout(4000)
    const webtext = await page.locator('#result').textContent()
    console.log(webtext)
    await expect(page.locator('#result')).toContainText(/You successfully clicked/)
    await page.waitForTimeout(4000)

})

test('Handling confirm',async({page})=>{
    await page.goto('https://the-internet.herokuapp.com/javascript_alerts')
    await page.waitForTimeout(3000)
    page.on('dialog',async(dialog)=>{
        const alerttype = dialog.type()
        console.log(`Alert type:: ${alerttype}`)
        const alertmessage = dialog.message()
        console.log(`Alert message is:: ${alertmessage}`)
        await page.waitForTimeout(4000)
        dialog.dismiss()
    })

    await page.getByText('Click for JS Confirm').click()
    const webtext = await page.locator('#result').innerText()
    console.log(webtext)
    await expect(page.locator('#result')).toBeVisible()
 await page.waitForTimeout(4000)
})
